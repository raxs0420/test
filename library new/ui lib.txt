-- Complete UI Library for TDS - Grey/Green Edition
local UILib = {}

-- Cache screen size for performance
local camera = workspace.CurrentCamera
local UIS = game:GetService("UserInputService")

-- Helper functions
local screenSize = workspace.CurrentCamera.ViewportSize
local UI_SCALE = 0.75 -- 25% smaller (1 = 100%, 0.75 = 75%)

local function scaleX(percent)
    return screenSize.X * (percent / 100) * UI_SCALE
end

local function scaleY(percent)
    return screenSize.Y * (percent / 100) * UI_SCALE
end

-- ========== CORE METHODS ==========
function UILib:AddCorner(instance, scale)
    if not instance then return end
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, scaleX(scale or self.Config.Sizes.CornerRadius))
    corner.Parent = instance
end

function UILib:AddStroke(instance, thickness, color)
    if not instance then return end
    local stroke = Instance.new("UIStroke")
    stroke.Thickness = thickness or 1
    stroke.Color = color or self.Config.Colors.Border
    stroke.Transparency = self.Config.Transparency.Borders
    stroke.Parent = instance
end

function UILib:CreateScreenGui(name, parent)
    local gui = Instance.new("ScreenGui")
    gui.Name = name or "UILibGUI"
    gui.ResetOnSpawn = false
    gui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    gui.Parent = parent or game:GetService("Players").LocalPlayer:WaitForChild("PlayerGui")
    return gui
end

-- ========== WINDOW ==========
function UILib:CreateWindow(parent, title, size, position, draggable)
    size = size or UDim2.new(0, scaleX(42), 0, scaleY(70))
    position = position or UDim2.new(0, scaleX(7), 0.5, -scaleY(87)/2)
    
    local window = Instance.new("Frame")
    window.Size = size
    window.Position = position
    window.BackgroundColor3 = self.Config.Colors.Background
    window.BackgroundTransparency = 0.2
    window.BorderSizePixel = 0
    window.Parent = parent
    window.Active = true
    window.ClipsDescendants = false
    
    self:AddCorner(window, self.Config.Sizes.LargeCorner)
    self:AddStroke(window, self.Config.Sizes.BorderSize, self.Config.Colors.BorderLight)
    
    return window
end

function UILib:CreateMinimizeButton(parent, window)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0, scaleX(4.5), 0, scaleY(5.5))
    btn.Position = UDim2.new(0, scaleX(1), 0, scaleY(1))
    btn.BackgroundColor3 = self.Config.Colors.Accent
    btn.Text = "−"
    btn.TextColor3 = self.Config.Colors.Text
    btn.Font = self.Config.Fonts.Bold
    btn.TextSize = scaleY(2.5)
    btn.Parent = parent
    btn.AutoButtonColor = false
    
    self:AddCorner(btn, 0.5)
    self:AddStroke(btn, self.Config.Sizes.BorderSize, self.Config.Colors.Border)

    btn.MouseButton1Click:Connect(function()
        minimized = not minimized
        window.Visible = not minimized
        btn.Text = minimized and "□" or "−"
    end)
    
    return btn
end

-- ========== DRAGGABLE ==========
function UILib:MakeDraggable(frame, handle)
    if not frame then return end
    handle = handle or frame
    
    local dragging = false
    local dragStart
    local startPos
    
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
        end
    end)
    
    handle.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    
    UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

-- ========== BASIC ELEMENTS ==========
function UILib:CreateFrame(parent, size, position, transparent, color)
    if not parent then return nil end
    local frame = Instance.new("Frame")
    frame.Size = size or UDim2.new(1, 0, 1, 0)
    frame.Position = position or UDim2.new(0, 0, 0, 0)
    frame.BackgroundTransparency = transparent and 1 or 0.2
    frame.BackgroundColor3 = color or self.Config.Colors.Background
    frame.BorderSizePixel = 0
    frame.Parent = parent
    return frame
end

function UILib:CreateLabel(parent, text, size, position, color, textSize, isBold, alignment)
    if not parent then return nil end
    local label = Instance.new("TextLabel")
    label.Size = size or UDim2.new(1, 0, 0, scaleY(4))
    label.Position = position or UDim2.new(0, 0, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = color or self.Config.Colors.Text
    label.Font = isBold and self.Config.Fonts.Bold or self.Config.Fonts.Normal
    label.TextSize = textSize or scaleY(1.4)
    label.TextXAlignment = alignment or Enum.TextXAlignment.Center
    label.Parent = parent
    return label
end

-- ========== BUTTON ==========
function UILib:CreateButton(parent, text, size, position, color, callback)
    if not parent then return nil end
    
    local btn = Instance.new("TextButton")
    btn.Size = size or UDim2.new(0, scaleX(10), 0, scaleY(5))
    btn.Position = position or UDim2.new(0, 0, 0, 0)
    btn.BackgroundColor3 = color or self.Config.Colors.Accent
    btn.Text = text or "Button"
    btn.TextColor3 = self.Config.Colors.Text
    btn.Font = self.Config.Fonts.Bold
    btn.TextSize = scaleY(1.6)
    btn.Parent = parent
    btn.AutoButtonColor = false
    
    self:AddCorner(btn)
    self:AddStroke(btn, self.Config.Sizes.BorderSize, self.Config.Colors.Border)
    
    if callback then
        btn.MouseButton1Click:Connect(function()
            local success, err = pcall(callback)
            if not success then
                warn("Button error:", err)
            end
        end)
    end
    
    return btn
end

    box.ClearTextOnFocus = false
    
    self:AddCorner(box)
    self:AddStroke(box, self.Config.Sizes.BorderSize, self.Config.Colors.Border)
    
    return box
end

-- ========== TOWER BUTTON ==========
function UILib:CreateTowerButton(parent, towerName, index, isSelected, callback)
    if not parent then return nil end
    
    local isEven = index % 2 == 0
    local baseColor = isEven and Color3.fromRGB(40, 40, 48) or Color3.fromRGB(35, 35, 42)
    
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, -scaleX(0.8), 0, scaleY(5.5))
    btn.Position = UDim2.new(0, scaleX(0.4), 0, (index-1) * scaleY(6))
    btn.BackgroundColor3 = isSelected and self.Config.Colors.Selected or baseColor
    btn.Text = towerName
    btn.TextColor3 = self.Config.Colors.Text
    btn.TextSize = scaleY(1.6)
    btn.Font = self.Config.Fonts.Normal
    btn.Parent = parent
    btn.AutoButtonColor = false
    
    self:AddCorner(btn, 0.3)
    self:AddStroke(btn, self.Config.Sizes.BorderSize, self.Config.Colors.Border)
    
    if callback then
        btn.MouseButton1Click:Connect(function()
            local success, err = pcall(function() callback(btn, towerName, baseColor) end)
            if not success then warn("Tower button error:", err) end
        end)
    end
    
    return btn
end

-- ========== MINI TOGGLE ROW ==========
function UILib:CreateMiniToggleRow(parent, buttons)
    if not parent then return nil, {} end
    
    local row = Instance.new("Frame")
    row.Size = UDim2.new(1, -scaleX(2), 0, scaleY(6))
    row.Position = UDim2.new(0, scaleX(1), 0, 0)
    row.BackgroundTransparency = 1
    row.Parent = parent
    
    local toggles = {}
    local buttonWidth = 0.19
    
    for i, data in ipairs(buttons) do
        local xPos = buttonWidth * (i-1) + (0.01 * (i-1))
        local btn, state = self:CreateToggleButton(row, data.name, data.default,
            UDim2.new(buttonWidth, 0, 1, 0),
            UDim2.new(xPos, 0, 0, 0),
            data.callback
        )
        toggles[data.name] = {button = btn, state = state}
    end
    
    return row, toggles
end

-- ========== TOGGLE BUTTON ==========
function UILib:CreateToggleButton(parent, text, defaultState, size, position, callback)
    if not parent then return nil, defaultState end
    
    local btn = Instance.new("TextButton")
    btn.Size = size or UDim2.new(0, scaleX(8), 0, scaleY(4))
    btn.Position = position or UDim2.new(0, 0, 0, 0)
    btn.BackgroundColor3 = defaultState and self.Config.Colors.Success or self.Config.Colors.BackgroundLight
    btn.Text = text .. (defaultState and " ✓" or " ✗")
    btn.TextColor3 = self.Config.Colors.Text
    btn.Font = self.Config.Fonts.Bold
    btn.TextSize = scaleY(1.4)
    btn.Parent = parent
    btn.AutoButtonColor = false
    
    self:AddCorner(btn)
    self:AddStroke(btn, self.Config.Sizes.BorderSize, self.Config.Colors.Border)
    
    local state = defaultState
    local originalColor = btn.BackgroundColor3
    
    btn.MouseButton1Click:Connect(function()
        state = not state
        btn.BackgroundColor3 = state and self.Config.Colors.Success or self.Config.Colors.BackgroundLight
        btn.Text = text .. (state and " ✓" or " ✗")
        if callback then
            local success, err = pcall(function() callback(state) end)
            if not success then warn("Toggle error:", err) end
        end
    end)
    
    return btn, state
end

-- ========== CLEAR CONTAINER ==========
function UILib:ClearContainer(container)
    if not container then return end
    for _, child in ipairs(container:GetChildren()) do
        child:Destroy()
    end
end

return UILib
