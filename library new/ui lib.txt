-- Complete UI Library for TDS - Grey/Green Edition
local UILib = {}

-- Cache screen size for performance
local camera = workspace.CurrentCamera
local UIS = game:GetService("UserInputService")

-- Helper functions
local function scaleX(percent)
    return camera.ViewportSize.X * (percent / 100)
end

local function scaleY(percent)
    -- Increase base size by 20% and ensure minimum for text clarity
    local calculated = camera.ViewportSize.Y * (percent / 100) * 1.1
    local minSize = 14  -- Absolute minimum pixel size
    return math.max(calculated, minSize)
end

-- ========== TRANSPARENCY FUNCTIONS ==========

-- Set transparency for a single element
function UILib:SetTransparency(element, transparency)
    if not element then return end
    element.BackgroundTransparency = transparency
end

-- Set text transparency
function UILib:SetTextTransparency(element, transparency)
    if not element then return end
    if element:IsA("TextLabel") or element:IsA("TextButton") or element:IsA("TextBox") then
        element.TextTransparency = transparency
    end
end

-- Create a subtle gradient fade (optional, not see-through)
function UILib:AddVerticalGradient(element, startTransparency, endTransparency)
    local gradient = Instance.new("UIGradient")
    gradient.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 255, 255)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 255, 255))
    })
    gradient.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, startTransparency),
        NumberSequenceKeypoint.new(1, endTransparency)
    })
    gradient.Rotation = 90  -- Vertical gradient
    gradient.Parent = element
    return gradient
end

-- ========== UI CREATION METHODS ==========

-- Core methods
function UILib:AddCorner(instance, scale)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, scaleX(scale or self.Config.Sizes.CornerRadius))
    corner.Parent = instance
end

function UILib:AddStroke(instance, thickness, color)
    local stroke = Instance.new("UIStroke")
    stroke.Thickness = thickness or 1
    stroke.Color = color or self.Config.Colors.Border
    stroke.Transparency = self.Config.Transparency.Borders
    stroke.Parent = instance
end

function UILib:CreateScreenGui(name, parent)
    local gui = Instance.new("ScreenGui")
    gui.Name = name or "UILibGUI"
    gui.ResetOnSpawn = false
    gui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    gui.Parent = parent or game:GetService("Players").LocalPlayer:WaitForChild("PlayerGui")
    return gui
end

function UILib:CreateWindow(parent, title, size, position, draggable)
    size = size or UDim2.new(0, scaleX(42), 0, scaleY(70))
    position = position or UDim2.new(0, scaleX(7), 0.5, -scaleY(87)/2)
    
    local window = Instance.new("Frame")
    window.Size = size
    window.Position = position
    window.BackgroundColor3 = self.Config.Colors.Background
    window.BackgroundTransparency = self.Config.Transparency.Background
    window.BorderSizePixel = 0
    window.Parent = parent
    
    self:AddCorner(window, self.Config.Sizes.LargeCorner)
    self:AddStroke(window, self.Config.Sizes.BorderSize, self.Config.Colors.BorderLight)
    
    if draggable then
        self:MakeDraggable(window, window)
    end
    
    return window
end

function UILib:CreateMinimizeButton(parent, window, size, position)
    local btn = Instance.new("TextButton")
    btn.Size = size or UDim2.new(0, scaleX(4.5), 0, scaleY(5.5))
    btn.Position = position or UDim2.new(0, scaleX(2), 0, scaleY(2))
    btn.BackgroundColor3 = self.Config.Colors.BackgroundLight
    btn.BackgroundTransparency = self.Config.Transparency.Buttons
    btn.BorderSizePixel = 0
    btn.Text = "▼"
    btn.TextColor3 = self.Config.Colors.Text
    btn.TextTransparency = self.Config.Transparency.Text
    btn.Font = self.Config.Fonts.Bold
    btn.TextSize = scaleY(1.8)
    btn.Parent = parent
    
    self:AddCorner(btn, 0.5)
    self:AddStroke(btn, self.Config.Sizes.BorderSize, self.Config.Colors.Border)
    
    local minimized = false
    local baseTransparency = self.Config.Transparency.Buttons
    
    -- Hover effects
    btn.MouseEnter:Connect(function()
        if not minimized then
            btn.BackgroundTransparency = math.max(baseTransparency - self.Config.Transparency.Hover, 0)
            btn.BackgroundColor3 = btn.BackgroundColor3:Lerp(self.Config.Colors.Accent, 0.3)
        end
    end)
    
    btn.MouseLeave:Connect(function()
        if not minimized then
            btn.BackgroundTransparency = baseTransparency
            btn.BackgroundColor3 = self.Config.Colors.BackgroundLight
        end
    end)
    
    btn.MouseButton1Click:Connect(function()
        minimized = not minimized
        window.Visible = not minimized
        btn.Text = minimized and "▲" or "▼"
        btn.BackgroundColor3 = minimized and self.Config.Colors.Accent or self.Config.Colors.BackgroundLight
    end)
    
    return btn
end

function UILib:CreateTab(parent, name, position, isActive)
    local tab = Instance.new("TextButton")
    tab.Size = UDim2.new(0.32, -scaleX(0.5), 1, -scaleY(1))
    tab.Position = position
    tab.BackgroundColor3 = isActive and self.Config.Colors.Accent or self.Config.Colors.BackgroundLight
    tab.BackgroundTransparency = isActive and self.Config.Transparency.Buttons or self.Config.Transparency.Elements
    tab.Text = name
    tab.TextColor3 = self.Config.Colors.Text
    tab.TextTransparency = self.Config.Transparency.Text
    tab.Font = self.Config.Fonts.Bold
    tab.TextSize = scaleY(1.6)
    tab.Parent = parent
    
    self:AddCorner(tab, 0.5)
    
    -- Hover effect for inactive tabs
    if not isActive then
        local baseTransparency = self.Config.Transparency.Elements
        tab.MouseEnter:Connect(function()
            tab.BackgroundTransparency = math.max(baseTransparency - self.Config.Transparency.Hover, 0)
            tab.BackgroundColor3 = tab.BackgroundColor3:Lerp(self.Config.Colors.Accent, 0.2)
        end)
        tab.MouseLeave:Connect(function()
            tab.BackgroundTransparency = baseTransparency
            tab.BackgroundColor3 = self.Config.Colors.BackgroundLight
        end)
    end
    
    return tab
end

function UILib:CreateFrame(parent, size, position, transparent, color)
    local frame = Instance.new("Frame")
    frame.Size = size or UDim2.new(1, 0, 1, 0)
    frame.Position = position or UDim2.new(0, 0, 0, 0)
    frame.BackgroundTransparency = transparent and 1 or (color and self.Config.Transparency.Elements or 0)
    frame.BackgroundColor3 = color or self.Config.Colors.Background
    frame.BorderSizePixel = 0
    frame.Parent = parent
    
    return frame
end

function UILib:CreateButton(parent, text, size, position, color, callback)
    local btn = Instance.new("TextButton")
    btn.Size = size or UDim2.new(0, scaleX(10), 0, scaleY(self.Config.Sizes.ButtonHeight))
    btn.Position = position or UDim2.new(0, 0, 0, 0)
    btn.BackgroundColor3 = color or self.Config.Colors.Accent
    btn.BackgroundTransparency = self.Config.Transparency.Buttons
    btn.Text = text
    btn.TextColor3 = self.Config.Colors.Text
    btn.TextTransparency = self.Config.Transparency.Text
    btn.Font = self.Config.Fonts.Bold
    btn.TextSize = scaleY(1.6)
    btn.Parent = parent
    
    self:AddCorner(btn)
    self:AddStroke(btn, self.Config.Sizes.BorderSize, self.Config.Colors.Border)
    
    local originalColor = btn.BackgroundColor3
    local baseTransparency = self.Config.Transparency.Buttons
    
    -- Hover effects
    btn.MouseEnter:Connect(function()
        btn.BackgroundTransparency = math.max(baseTransparency - self.Config.Transparency.Hover, 0)
        btn.BackgroundColor3 = originalColor:Lerp(self.Config.Colors.Text, 0.2)
    end)
    
    btn.MouseLeave:Connect(function()
        btn.BackgroundTransparency = baseTransparency
        btn.BackgroundColor3 = originalColor
    end)
    
    btn.MouseButton1Down:Connect(function()
        btn.BackgroundTransparency = math.min(baseTransparency + self.Config.Transparency.Pressed, 1)
        btn.BackgroundColor3 = originalColor:Lerp(self.Config.Colors.BackgroundDark, 0.3)
    end)
    
    btn.MouseButton1Up:Connect(function()
        btn.BackgroundTransparency = math.max(baseTransparency - self.Config.Transparency.Hover, 0)
        btn.BackgroundColor3 = originalColor:Lerp(self.Config.Colors.Text, 0.2)
    end)
    
    if callback then
        btn.MouseButton1Click:Connect(callback)
    end
    
    return btn
end

function UILib:CreateTextBox(parent, placeholder, size, position, text)
    local box = Instance.new("TextBox")
    box.Size = size or UDim2.new(1, -scaleX(4), 0, scaleY(self.Config.Sizes.TextBoxHeight))
    box.Position = position or UDim2.new(0, scaleX(2), 0, scaleY(2))
    box.BackgroundColor3 = self.Config.Colors.BackgroundDarker
    box.BackgroundTransparency = self.Config.Transparency.Elements
    box.TextColor3 = self.Config.Colors.Text
    box.TextTransparency = self.Config.Transparency.Text
    box.PlaceholderText = placeholder or "Search..."
    box.PlaceholderColor3 = self.Config.Colors.TextMuted
    box.Text = text or ""
    box.Font = self.Config.Fonts.Normal
    box.TextSize = scaleY(1.6)
    box.ClearTextOnFocus = false
    box.Parent = parent
    
    self:AddCorner(box)
    self:AddStroke(box, self.Config.Sizes.BorderSize, self.Config.Colors.Border)
    
    -- Focus effects
    box.Focused:Connect(function()
        self:AddStroke(box, self.Config.Sizes.BorderSize + 1, self.Config.Colors.Accent)
        box.BackgroundTransparency = math.max(self.Config.Transparency.Elements - 0.05, 0)
    end)
    
    box.FocusLost:Connect(function()
        self:AddStroke(box, self.Config.Sizes.BorderSize, self.Config.Colors.Border)
        box.BackgroundTransparency = self.Config.Transparency.Elements
    end)
    
    return box
end

function UILib:CreateSlotButton(parent, slotNumber, size, position, callback)
    local frame = Instance.new("Frame")
    frame.Size = size or UDim2.new(1, 0, 0, scaleY(7))
    frame.Position = position or UDim2.new(0, 0, 0, (slotNumber-1) * scaleY(7))
    frame.BackgroundTransparency = 1  -- Container frame is transparent
    frame.Parent = parent
    
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0.25, 0, 1, -scaleY(0.5))
    btn.Position = UDim2.new(0, 0, 0, scaleY(0.25))
    btn.BackgroundColor3 = self.Config.Colors.SlotEmpty
    btn.BackgroundTransparency = self.Config.Transparency.Buttons
    btn.BorderSizePixel = 0
    btn.Text = tostring(slotNumber)
    btn.TextColor3 = self.Config.Colors.Text
    btn.TextTransparency = self.Config.Transparency.Text
    btn.Font = self.Config.Fonts.Bold
    btn.TextSize = scaleY(1.8)
    btn.Parent = frame
    
    self:AddCorner(btn)
    self:AddStroke(btn, self.Config.Sizes.BorderSize, self.Config.Colors.Border)
    
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.75, -scaleX(1), 1, -scaleY(0.5))
    label.Position = UDim2.new(0.25, scaleX(1), 0, scaleY(0.25))
    label.BackgroundColor3 = self.Config.Colors.BackgroundLight
    label.BackgroundTransparency = self.Config.Transparency.Elements
    label.TextColor3 = self.Config.Colors.Text
    label.TextTransparency = self.Config.Transparency.Text
    label.Font = self.Config.Fonts.Normal
    label.TextSize = scaleY(1.4)
    label.Text = "[Empty]"
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.TextYAlignment = Enum.TextYAlignment.Center
    label.Parent = frame
    
    self:AddCorner(label)
    self:AddStroke(label, self.Config.Sizes.BorderSize, self.Config.Colors.Border)
    
    -- Hover effects with transparency
    local baseTransparency = self.Config.Transparency.Buttons
    btn.MouseEnter:Connect(function()
        if btn.BackgroundColor3 ~= self.Config.Colors.Selected and 
           btn.BackgroundColor3 ~= self.Config.Colors.SlotFilled then
            btn.BackgroundTransparency = math.max(baseTransparency - self.Config.Transparency.Hover, 0)
            btn.BackgroundColor3 = btn.BackgroundColor3:Lerp(self.Config.Colors.AccentLight, 0.3)
        end
    end)
    
    btn.MouseLeave:Connect(function()
        if btn.BackgroundColor3 ~= self.Config.Colors.Selected and 
           btn.BackgroundColor3 ~= self.Config.Colors.SlotFilled then
            btn.BackgroundTransparency = baseTransparency
            btn.BackgroundColor3 = self.Config.Colors.SlotEmpty
        end
    end)
    
    if callback then
        btn.MouseButton1Click:Connect(function()
            callback(slotNumber, btn, label)
        end)
    end
    
    return btn, label, frame
end

function UILib:CreateToggleButton(parent, text, defaultState, size, position, callback)
    local btn = Instance.new("TextButton")
    btn.Size = size or UDim2.new(0, scaleX(8), 0, scaleY(4))
    btn.Position = position or UDim2.new(0, 0, 0, 0)
    btn.BackgroundColor3 = defaultState and self.Config.Colors.Success or self.Config.Colors.BackgroundLight
    btn.BackgroundTransparency = self.Config.Transparency.Buttons
    btn.Text = text .. (defaultState and " ✓" or " ✗")
    btn.TextColor3 = self.Config.Colors.Text
    btn.TextTransparency = self.Config.Transparency.Text
    btn.Font = self.Config.Fonts.Bold
    btn.TextSize = scaleY(1.4)
    btn.Parent = parent
    
    self:AddCorner(btn)
    self:AddStroke(btn, self.Config.Sizes.BorderSize, self.Config.Colors.Border)
    
    local state = defaultState
    local originalColor = btn.BackgroundColor3
    local baseTransparency = self.Config.Transparency.Buttons
    
    btn.MouseEnter:Connect(function()
        btn.BackgroundTransparency = math.max(baseTransparency - self.Config.Transparency.Hover, 0)
        btn.BackgroundColor3 = originalColor:Lerp(self.Config.Colors.Text, 0.2)
    end)
    
    btn.MouseLeave:Connect(function()
        btn.BackgroundTransparency = baseTransparency
        btn.BackgroundColor3 = originalColor
    end)
    
    btn.MouseButton1Click:Connect(function()
        state = not state
        originalColor = state and self.Config.Colors.Success or self.Config.Colors.BackgroundLight
        btn.BackgroundColor3 = originalColor
        btn.Text = text .. (state and " ✓" or " ✗")
        if callback then
            callback(state)
        end
    end)
    
    return btn, state
end

function UILib:CreateMiniToggleRow(parent, buttons, startX)
    local row = Instance.new("Frame")
    row.Size = UDim2.new(1, -scaleX(2), 0, scaleY(6))
    row.Position = UDim2.new(0, scaleX(1), 0, 0)
    row.BackgroundTransparency = 1
    row.Parent = parent
    
    local buttonWidth = 0.192
    local buttonSpacing = 0.01
    local toggles = {}
    
    for i, btnData in ipairs(buttons) do
        local xPos = (buttonWidth + buttonSpacing) * (i-1)
        local toggle, state = self:CreateToggleButton(row, btnData.name, btnData.default,
            UDim2.new(buttonWidth, 0, 1, 0),
            UDim2.new(xPos, 0, 0, 0),
            btnData.callback
        )
        toggle.TextSize = scaleY(0.9)
        toggles[btnData.name] = {button = toggle, state = state}
    end
    
    return row, toggles
end

function UILib:CreateConfirmationPopup(parent, title, message, onConfirm, onCancel)

        scaleY(1.4),

    )

        UDim2.new(1, -scaleX(2), 0, scaleY(5)),
        UDim2.new(0, scaleX(1), 0, scaleY(10)),

function UILib:MakeDraggable(frame, handle)
    local dragging = false
    local dragStart
    local startPos
    
    handle = handle or frame
    
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or 
           input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
        end
    end)
    
    handle.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or 
           input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    
    UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or 
                        input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            local newX = startPos.X.Offset + delta.X
            local newY = startPos.Y.Offset + delta.Y
            
            local screenBounds = camera.ViewportSize
            local frameSize = frame.AbsoluteSize
            
            newX = math.max(0, math.min(newX, screenBounds.X - frameSize.X))
            newY = math.max(0, math.min(newY, screenBounds.Y - frameSize.Y))
            
            frame.Position = UDim2.new(0, newX, 0, newY)
        end
    end)
end

function UILib:ClearContainer(container)
    for _, child in ipairs(container:GetChildren()) do
        child:Destroy()
    end
end
