local function FindPath()
    local MapFolder = workspace:FindFirstChild("Map")
    if not MapFolder then return nil end
    local PathsFolder = MapFolder:FindFirstChild("Paths")
    if not PathsFolder then return nil end
    local PathFolder = PathsFolder:GetChildren()[1]
    if not PathFolder then return nil end
    
    local nodes = {}
    for _, node in ipairs(PathFolder:GetChildren()) do
        if node:IsA("BasePart") then
            table.insert(nodes, node)
        end
    end
    
    table.sort(nodes, function(a, b)
        local NumA = tonumber(a.Name:match("%d+"))
        local NumB = tonumber(b.Name:match("%d+"))
        if NumA and NumB then return NumA < NumB end
        return a.Name < b.Name
    end)
    
    return nodes
end

local function TotalLength(nodes)
    local total = 0
    for i = 1, #nodes - 1 do
        total = total + (nodes[i + 1].Position - nodes[i].Position).Magnitude
    end
    return total
end

local function GetPointAtDistance(nodes, distance)
    if not nodes or #nodes < 2 then return nil end
    
    local currentDist = 0
    for i = 1, #nodes - 1 do
        local startPos = nodes[i].Position
        local endPos = nodes[i + 1].Position
        local segmentLen = (endPos - startPos).Magnitude
        
        if currentDist + segmentLen >= distance then
            local remaining = distance - currentDist
            local direction = (endPos - startPos).Unit
            return startPos + (direction * remaining)
        end
        currentDist = currentDist + segmentLen
    end
    return nodes[#nodes].Position
end

local function UpdatePath()
    local nodes = FindPath()
    if nodes and #nodes > 0 then
        PathNodes = nodes
        MaxPathDistance = TotalLength(nodes)
        return true
    end
    return false
end

UpdatePath()

local function isDamageNumber(obj)
    if obj:IsA("BillboardGui") then
        return true
    end
    
    if obj:IsA("TextLabel") then
        if obj.Text and obj.Text:match("^%-?%d+") then
            return true
        end
    end
    
    return false
end

local function checkAndRemove(obj)
    if isDamageNumber(obj) then
        obj:Destroy()
    end
end

local function findGatlingGun()
    local towers = workspace:FindFirstChild("Towers")
    if not towers then return nil, nil end
    
    for _, towerFolder in pairs(towers:GetChildren()) do
        local rep = towerFolder:FindFirstChild("TowerReplicator")
        if rep then
            local ownerAttr = rep:GetAttribute("OwnerId")
            if ownerAttr and ownerAttr == LocalPlayer.UserId then
                local nameAttr = rep:GetAttribute("Name")
                if nameAttr and (nameAttr == "Gatlin Gun" or nameAttr == "Gatling Gun") then
                    return rep, towerFolder
                end
            end
        end
    end
    return nil, nil
end

local function getEnemyPathPosition(npc)
    if #PathNodes == 0 then return nil end
    
    local humanoid = npc:FindFirstChildOfClass("Humanoid")
    if not humanoid then return nil end
    
    local rootPart = npc:FindFirstChild("HumanoidRootPart") or npc.PrimaryPart
    if not rootPart then return nil end
    
    local closestDist = math.huge
    local closestPoint = nil
    local closestNodeIndex = 1
    
    for i = 1, #PathNodes - 1 do
        local startPos = PathNodes[i].Position
        local endPos = PathNodes[i + 1].Position
        local segmentDir = (endPos - startPos).Unit
        local toEnemy = rootPart.Position - startPos
        
        local t = toEnemy:Dot(segmentDir)
        t = math.max(0, math.min(t, (endPos - startPos).Magnitude))
        
        local pointOnSegment = startPos + segmentDir * t
        local dist = (rootPart.Position - pointOnSegment).Magnitude
        
        if dist < closestDist then
            closestDist = dist
            closestPoint = pointOnSegment
            closestNodeIndex = i + (t / (endPos - startPos).Magnitude)
        end
    end
    
    local pathDistance = 0
    for i = 1, closestNodeIndex - 1 do
        pathDistance = pathDistance + (PathNodes[i + 1].Position - PathNodes[i].Position).Magnitude
    end
    
    if closestNodeIndex > math.floor(closestNodeIndex) then
        local i = math.floor(closestNodeIndex)
        local t = closestNodeIndex - i
        pathDistance = pathDistance + t * (PathNodes[i + 1].Position - PathNodes[i].Position).Magnitude
    end
    
    return pathDistance
end

local function isEnemyMoving(npc)
    local humanoid = npc:FindFirstChildOfClass("Humanoid")
    if humanoid then
        if humanoid.MoveDirection.Magnitude > 0 then
            return true
        end
        
        local rootPart = npc:FindFirstChild("HumanoidRootPart") or npc.PrimaryPart
        if rootPart and rootPart:IsA("BasePart") and rootPart.Velocity.Magnitude > 1 then
            return true
        end
    end
    return false
end

local function getEnemies()
    local enemies = {}
    local movingEnemies = {}
    
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return enemies, movingEnemies end
    
    local npcs = workspace:FindFirstChild("NPCs")
    if not npcs then return enemies, movingEnemies end
    
    for _, npc in pairs(npcs:GetChildren()) do
        local humanoid = npc:FindFirstChildOfClass("Humanoid")
        if humanoid and humanoid.Health > 0 then
            local part = npc:FindFirstChild("Hitbox")
            
            if part then
                local dist = (root.Position - part.Position).Magnitude
                if dist <= 500 then
                    local moving = isEnemyMoving(npc)
                    local pathPos = getEnemyPathPosition(npc) or 0
                    
                    local enemyData = {
                        npc = npc,
                        part = part,
                        pos = part.Position,
                        dist = dist,
                        pathDist = pathPos,
                        isMoving = moving
                    }
                    
                    table.insert(enemies, enemyData)
                    
                    if moving then
                        table.insert(movingEnemies, enemyData)
                    end
                end
            end
        end
    end
    
    table.sort(enemies, function(a, b) return a.pathDist > b.pathDist end)
    table.sort(movingEnemies, function(a, b) return a.pathDist > b.pathDist end)
    
    Stats.Enemies = #enemies
    return enemies, movingEnemies
end

local function findEnemiesInLine(primaryEnemy, allEnemies)
    local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not root then return {primaryEnemy} end
    
    local camera = workspace.CurrentCamera
    local shootDir = (primaryEnemy.pos - camera.CFrame.Position).Unit
    local targets = {primaryEnemy}
    
    for _, enemy in ipairs(allEnemies) do
        if enemy ~= primaryEnemy and #targets < 3 then 
            local toEnemy = (enemy.pos - camera.CFrame.Position).Unit
            local dot = shootDir:Dot(toEnemy)
            
            if dot > 0.7 then
                local distToPrimary = (enemy.pos - primaryEnemy.pos).Magnitude
                if distToPrimary <= 25 then
                    table.insert(targets, enemy)
                end
            end
        end
    end
    
    return targets
end

local function selectTargets()
    local allEnemies, movingEnemies = getEnemies()
    
    local primaryList = #movingEnemies > 0 and movingEnemies or allEnemies
    if #primaryList == 0 then return {} end
    
    local primary = primaryList[1]
    
    local targets = findEnemiesInLine(primary, primaryList)
    
    return targets
end

local function stopFiring()
    firing = false
    if connection then 
        connection:Disconnect() 
        connection = nil 
    end
end

local function startFiring()
    if firing then return end
    firing = true
    
    connection = RunService.Heartbeat:Connect(function()
        if not firing or not Settings.On or not Stats.TowerFound or not Stats.CanFire or not GatlingGunRemote then
            return
        end
        
        local targets = selectTargets()
        if #targets == 0 then return end
        
        local sync = workspace:GetAttribute("Sync") or 0
        local time = workspace:GetServerTimeNow() or 0
        local camera = workspace.CurrentCamera
        
        if camera and targets[1] then
            local dir = (targets[1].pos - camera.CFrame.Position).Unit
            camera.CFrame = CFrame.lookAt(camera.CFrame.Position, camera.CFrame.Position + dir)
        end
        
        for _, target in ipairs(targets) do
            if not firing or not Stats.CanFire then break end
            
            for i = 1, Settings.Burst do
                if not firing or not Stats.CanFire then break end
                pcall(function() 
                    GatlingGunRemote:FireServer(target.pos, sync, time) 
                end)
                if i < Settings.Burst then task.wait(0.001) end
            end
        end
        task.wait(Settings.Cooldown)
    end)
end

spawn(function()
    while true do
        task.wait(0.1)
        if Stats.TowerFound and Stats.CanFire and GatlingGunRemote then
            if not firing then
                startFiring()
            end
        else
            if firing then
                stopFiring()
            end
        end
    end
end)
