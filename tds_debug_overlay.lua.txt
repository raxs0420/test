-- // TDS Debug Overlay - ULTRA SIMPLE GUARANTEED WORKING VERSION

local function attachOverlay(TDS)
    print("[DEBUG] attachOverlay called")
    
    if not TDS then 
        print("[DEBUG] TDS is nil, creating mock")
        TDS = {}
    end
    
    local Players = game:GetService("Players")
    local localPlayer = Players.LocalPlayer
    
    if not localPlayer then
        print("[DEBUG] No local player yet")
        return
    end
    
    print("[DEBUG] Local player found:", localPlayer.Name)
    
    local playerGui = localPlayer:FindFirstChild("PlayerGui")
    if not playerGui then
        print("[DEBUG] Waiting for PlayerGui...")
        playerGui = localPlayer:WaitForChild("PlayerGui", 5)
        if not playerGui then
            print("[DEBUG] Failed to get PlayerGui")
            return
        end
    end
    
    print("[DEBUG] PlayerGui found")
    
    -- SIMPLE WINDOW - no fancy stuff, just a basic frame
    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "TDS_Debug_Test"
    screenGui.ResetOnSpawn = false
    screenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    screenGui.Parent = playerGui
    
    print("[DEBUG] ScreenGui parented to PlayerGui, Enabled:", screenGui.Enabled)
    
    -- Create a BIG, OBVIOUS window
    local mainFrame = Instance.new("Frame")
    mainFrame.Name = "MainFrame"
    mainFrame.Size = UDim2.new(0, 400, 0, 300)
    mainFrame.Position = UDim2.new(0.5, -200, 0.5, -150)  -- Center of screen
    mainFrame.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
    mainFrame.BackgroundTransparency = 0
    mainFrame.BorderSizePixel = 2
    mainFrame.BorderColor3 = Color3.fromRGB(255, 0, 0)
    mainFrame.Visible = true
    mainFrame.Active = true
    mainFrame.Parent = screenGui
    
    print("[DEBUG] Main frame created at center of screen")
    
    -- Add a title
    local title = Instance.new("TextLabel")
    title.Text = "TDS DEBUG OVERLAY - WORKING?"
    title.Size = UDim2.new(1, 0, 0, 40)
    title.Position = UDim2.new(0, 0, 0, 0)
    title.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
    title.TextColor3 = Color3.fromRGB(255, 255, 255)
    title.Font = Enum.Font.GothamBold
    title.TextSize = 16
    title.Parent = mainFrame
    
    -- Add a status label
    local status = Instance.new("TextLabel")
    status.Text = "Overlay is ACTIVE"
    status.Size = UDim2.new(1, 0, 0, 30)
    status.Position = UDim2.new(0, 0, 0, 45)
    status.BackgroundColor3 = Color3.fromRGB(0, 255, 0)
    status.TextColor3 = Color3.fromRGB(0, 0, 0)
    status.Font = Enum.Font.GothamBold
    status.TextSize = 14
    status.Parent = mainFrame
    
    -- Add a list for actions
    local actionList = Instance.new("ScrollingFrame")
    actionList.Size = UDim2.new(1, -10, 1, -90)
    actionList.Position = UDim2.new(0, 5, 0, 80)
    actionList.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    actionList.BorderSizePixel = 1
    actionList.BorderColor3 = Color3.fromRGB(100, 100, 100)
    actionList.CanvasSize = UDim2.new(0, 0, 0, 0)
    actionList.AutomaticCanvasSize = Enum.AutomaticSize.Y
    actionList.ScrollBarThickness = 6
    actionList.Parent = mainFrame
    
    local actionLayout = Instance.new("UIListLayout")
    actionLayout.Padding = UDim.new(0, 2)
    actionLayout.Parent = actionList
    
    local queue = {}
    
    -- Function to add an action
    local function addAction(text, statusType)
        print("[DEBUG] Adding action:", text, statusType)
        
        local row = Instance.new("Frame")
        row.Size = UDim2.new(1, 0, 0, 25)
        row.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
        row.BorderSizePixel = 0
        row.Parent = actionList
        
        local label = Instance.new("TextLabel")
        label.Text = text
        label.Size = UDim2.new(1, -60, 1, 0)
        label.Position = UDim2.new(0, 5, 0, 0)
        label.BackgroundTransparency = 1
        label.TextColor3 = Color3.fromRGB(255, 255, 255)
        label.TextXAlignment = Enum.TextXAlignment.Left
        label.Font = Enum.Font.Code
        label.TextSize = 11
        label.Parent = row
        
        local statusLabel = Instance.new("TextLabel")
        
        if statusType == "pending" then
            statusLabel.Text = "PENDING"
            statusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
        elseif statusType == "running" then
            statusLabel.Text = "RUNNING"
            statusLabel.TextColor3 = Color3.fromRGB(255, 255, 100)
        elseif statusType == "done" then
            statusLabel.Text = "DONE"
            statusLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
        elseif statusType == "error" then
            statusLabel.Text = "ERROR"
            statusLabel.TextColor3 = Color3.fromRGB(255, 50, 50)
        else
            statusLabel.Text = "ACTIVE"
            statusLabel.TextColor3 = Color3.fromRGB(100, 100, 255)
        end
        
        statusLabel.Size = UDim2.new(0, 55, 1, 0)
        statusLabel.Position = UDim2.new(1, -60, 0, 0)
        statusLabel.BackgroundTransparency = 1
        statusLabel.TextXAlignment = Enum.TextXAlignment.Right
        statusLabel.Font = Enum.Font.GothamBold
        statusLabel.TextSize = 10
        statusLabel.Parent = row
        
        local entry = {row = row, label = label, statusLabel = statusLabel, status = statusType}
        table.insert(queue, entry)
        
        -- Update counter
        local total = #queue
        local done = 0
        for _, e in ipairs(queue) do
            if e.status == "done" then done = done + 1 end
        end
        status.Text = "Actions: " .. done .. " / " .. total
        
        -- Auto-scroll
        task.defer(function()
            actionList.CanvasPosition = Vector2.new(0, actionList.CanvasSize.Y.Offset)
        end)
        
        return entry
    end
    
    -- Function to update status
    local function updateStatus(entry, newStatus)
        entry.status = newStatus
        if newStatus == "done" then
            entry.statusLabel.Text = "DONE"
            entry.statusLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
        elseif newStatus == "error" then
            entry.statusLabel.Text = "ERROR"
            entry.statusLabel.TextColor3 = Color3.fromRGB(255, 50, 50)
            entry.label.TextColor3 = Color3.fromRGB(255, 100, 100)
        elseif newStatus == "running" then
            entry.statusLabel.Text = "RUNNING"
            entry.statusLabel.TextColor3 = Color3.fromRGB(255, 255, 100)
        end
        
        -- Update counter
        local total = #queue
        local done = 0
        for _, e in ipairs(queue) do
            if e.status == "done" then done = done + 1 end
        end
        status.Text = "Actions: " .. done .. " / " .. total
    end
    
    -- Add close button
    local closeBtn = Instance.new("TextButton")
    closeBtn.Text = "X"
    closeBtn.Size = UDim2.new(0, 30, 0, 30)
    closeBtn.Position = UDim2.new(1, -35, 0, 5)
    closeBtn.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
    closeBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    closeBtn.Font = Enum.Font.GothamBold
    closeBtn.TextSize = 14
    closeBtn.Parent = mainFrame
    closeBtn.MouseButton1Click:Connect(function()
        screenGui:Destroy()
        print("[DEBUG] Overlay closed")
    end)
    
    -- Add minimize button
    local minBtn = Instance.new("TextButton")
    minBtn.Text = "-"
    minBtn.Size = UDim2.new(0, 30, 0, 30)
    minBtn.Position = UDim2.new(1, -70, 0, 5)
    minBtn.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
    minBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    minBtn.Font = Enum.Font.GothamBold
    minBtn.TextSize = 18
    minBtn.Parent = mainFrame
    
    local minimized = false
    minBtn.MouseButton1Click:Connect(function()
        minimized = not minimized
        if minimized then
            mainFrame.Size = UDim2.new(0, 400, 0, 45)
            actionList.Visible = false
            minBtn.Text = "+"
        else
            mainFrame.Size = UDim2.new(0, 400, 0, 300)
            actionList.Visible = true
            minBtn.Text = "-"
        end
    end)
    
    -- Add drag functionality
    local dragging = false
    local dragStart = nil
    local startPos = nil
    
    title.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPos = mainFrame.Position
        end
    end)
    
    game:GetService("UserInputService").InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - dragStart
            mainFrame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
    
    game:GetService("UserInputService").InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
    
    -- Wrap TDS methods
    local function wrapMethod(name, labelFunc)
        local original = TDS[name]
        if not original then
            -- Create mock method
            TDS[name] = function(self, ...)
                local args = {...}
                local label = labelFunc and labelFunc(table.unpack(args)) or name .. "()"
                local entry = addAction(label, "running")
                task.wait(0.2)
                updateStatus(entry, "done")
                return nil
            end
            print("[DEBUG] Created mock method:", name)
            return
        end
        
        TDS[name] = function(self, ...)
            local args = {...}
            local label = labelFunc(table.unpack(args))
            local entry = addAction(label, "running")
            
            local success, result = pcall(original, self, table.unpack(args))
            
            if success then
                updateStatus(entry, "done")
            else
                updateStatus(entry, "error")
                warn("[Overlay] Error in " .. name .. ": " .. tostring(result))
            end
            
            return result
        end
    end
    
    -- Wrap all methods
    wrapMethod("Mode", function(m) return "Mode: " .. tostring(m) end)
    wrapMethod("Loadout", function(...) 
        local args = {...}
        if #args == 0 then return "Loadout: (none)" end
        if #args == 1 then return "Loadout: " .. tostring(args[1]) end
        return "Loadout: " .. args[1] .. " +" .. (#args - 1) .. " more"
    end)
    wrapMethod("Ready", function() return "Ready() - waiting for wave" end)
    wrapMethod("Place", function(n, x, y, z) 
        return string.format("Place: %s @ (%.0f, %.0f)", tostring(n), x or 0, z or 0)
    end)
    wrapMethod("Upgrade", function(slot) return "Upgrade slot [" .. (slot or 0) .. "]" end)
    wrapMethod("Sell", function(slot) return "Sell slot [" .. (slot or 0) .. "]" end)
    wrapMethod("Ability", function(slot, name) 
        return "Ability: [" .. (slot or 0) .. "] " .. tostring(name)
    end)
    
    -- Add welcome actions
    addAction("✓ Overlay successfully loaded!", "done")
    addAction("→ Ready to monitor TDS calls", "pending")
    
    print("[DEBUG] Overlay setup complete! Window should be visible in the center of your screen")
    print("[DEBUG] ScreenGui parent:", screenGui.Parent)
    print("[DEBUG] MainFrame position:", mainFrame.AbsolutePosition)
    print("[DEBUG] MainFrame size:", mainFrame.AbsoluteSize)
    
    -- Force visibility check
    task.wait(0.5)
    mainFrame.Visible = true
    screenGui.Enabled = true
    
    return TDS
end

-- EXECUTION
print("[TDS Overlay] Starting...")

-- Check if TDS exists
local existingTDS = _G.TDS or TDS or nil

if existingTDS then
    print("[TDS Overlay] Found existing TDS, attaching...")
    attachOverlay(existingTDS)
else
    print("[TDS Overlay] No TDS found, creating test mode...")
    local testTDS = {}
    attachOverlay(testTDS)
    
    -- Run demo after 2 seconds
    task.spawn(function()
        wait(2)
        print("[TDS Overlay] Running demo...")
        testTDS:Mode("Test")
        wait(0.5)
        testTDS:Place("Demo Tower", 10, 0, 20)
        wait(0.5)
        testTDS:Upgrade(1)
        wait(0.5)
        testTDS:Loadout("Farm", "Militant")
        wait(0.5)
        testTDS:Ready()
    end)
end

print("[TDS Overlay] Script finished execution")
