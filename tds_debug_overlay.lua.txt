-- // TDS Debug Overlay
-- // GitHub: https://raw.githubusercontent.com/yourusername/yourrepo/main/tds_debug_overlay.lua
-- //
-- // USAGE:
-- //   local TDS = loadstring(game:HttpGet("your_tds_library_url"))()
-- //   loadstring(game:HttpGet("https://raw.githubusercontent.com/yourusername/yourrepo/main/tds_debug_overlay.lua"))()(TDS)

print("[TDS Overlay] Loading...")

local function attachOverlay(TDS)
    if not TDS then
        warn("[TDS Overlay] Error: TDS table is nil")
        return nil
    end
    
    print("[TDS Overlay] Attaching to TDS...")
    
    local Players = game:GetService("Players")
    local UIS = game:GetService("UserInputService")
    local TweenService = game:GetService("TweenService")
    local RunService = game:GetService("RunService")
    local localPlayer = Players.LocalPlayer
    local playerGui = localPlayer:WaitForChild("PlayerGui")
    
    -- Remove old overlay
    local existingOverlay = playerGui:FindFirstChild("TDS_DebugOverlay")
    if existingOverlay then
        existingOverlay:Destroy()
    end
    
    -- Create GUI
    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "TDS_DebugOverlay"
    screenGui.ResetOnSpawn = false
    screenGui.DisplayOrder = 999
    screenGui.Parent = playerGui
    
    -- Track slot information (which towers are in which slots)
    local slotTowers = {}
    
    -- Main window (initially minimized as small cube)
    local mainFrame = Instance.new("Frame")
    mainFrame.Size = UDim2.new(0, 80, 0, 80)
    mainFrame.Position = UDim2.new(0, 10, 0, 10)
    mainFrame.BackgroundColor3 = Color3.fromRGB(28, 28, 35)
    mainFrame.BorderSizePixel = 1
    mainFrame.BorderColor3 = Color3.fromRGB(55, 55, 75)
    mainFrame.Parent = screenGui
    
    local corner = Instance.new("UICorner", mainFrame)
    corner.CornerRadius = UDim.new(0, 8)
    
    -- Minimized button (cube)
    local minimizedButton = Instance.new("TextButton")
    minimizedButton.Size = UDim2.new(1, 0, 1, 0)
    minimizedButton.BackgroundTransparency = 1
    minimizedButton.Text = "Debug"
    minimizedButton.TextColor3 = Color3.fromRGB(225, 225, 245)
    minimizedButton.Font = Enum.Font.GothamBold
    minimizedButton.TextSize = 14
    minimizedButton.Parent = mainFrame
    
    -- Expanded window content (hidden initially)
    local expandedFrame = Instance.new("Frame")
    expandedFrame.Size = UDim2.new(0, 450, 0, 400)
    expandedFrame.Position = UDim2.new(0, 0, 0, 0)
    expandedFrame.BackgroundColor3 = Color3.fromRGB(18, 18, 23)
    expandedFrame.BorderSizePixel = 1
    expandedFrame.BorderColor3 = Color3.fromRGB(55, 55, 75)
    expandedFrame.Visible = false
    expandedFrame.Parent = mainFrame
    
    local expandedCorner = Instance.new("UICorner", expandedFrame)
    expandedCorner.CornerRadius = UDim.new(0, 8)
    
    -- Title bar
    local titleBar = Instance.new("Frame")
    titleBar.Size = UDim2.new(1, 0, 0, 40)
    titleBar.BackgroundColor3 = Color3.fromRGB(28, 28, 35)
    titleBar.BorderSizePixel = 0
    titleBar.Parent = expandedFrame
    
    local titleCorner = Instance.new("UICorner", titleBar)
    titleCorner.CornerRadius = UDim.new(0, 8)
    
    local titleText = Instance.new("TextLabel")
    titleText.Text = "TDS Debug"
    titleText.Size = UDim2.new(1, -60, 1, 0)
    titleText.Position = UDim2.new(0, 12, 0, 0)
    titleText.BackgroundTransparency = 1
    titleText.TextColor3 = Color3.fromRGB(225, 225, 245)
    titleText.TextXAlignment = Enum.TextXAlignment.Left
    titleText.Font = Enum.Font.GothamBold
    titleText.TextSize = 13
    titleText.Parent = titleBar
    
    -- Minimize button (to go back to cube)
    local minimizeBtn = Instance.new("TextButton")
    minimizeBtn.Text = "−"
    minimizeBtn.Size = UDim2.new(0, 30, 0, 30)
    minimizeBtn.Position = UDim2.new(1, -40, 0, 5)
    minimizeBtn.BackgroundColor3 = Color3.fromRGB(45, 45, 55)
    minimizeBtn.TextColor3 = Color3.fromRGB(190, 190, 215)
    minimizeBtn.Font = Enum.Font.GothamBold
    minimizeBtn.TextSize = 20
    minimizeBtn.BorderSizePixel = 0
    minimizeBtn.Parent = titleBar
    
    local minCorner = Instance.new("UICorner", minimizeBtn)
    minCorner.CornerRadius = UDim.new(0, 5)
    
    -- Scroll frame
    local scrollFrame = Instance.new("ScrollingFrame")
    scrollFrame.Size = UDim2.new(1, -12, 1, -50)
    scrollFrame.Position = UDim2.new(0, 6, 0, 46)
    scrollFrame.BackgroundColor3 = Color3.fromRGB(13, 13, 18)
    scrollFrame.BorderSizePixel = 0
    scrollFrame.CanvasSize = UDim2.new(0, 0, 0, 0)
    scrollFrame.ScrollBarThickness = 4
    scrollFrame.ScrollBarImageColor3 = Color3.fromRGB(70, 70, 95)
    scrollFrame.Parent = expandedFrame
    
    local scrollCorner = Instance.new("UICorner", scrollFrame)
    scrollCorner.CornerRadius = UDim.new(0, 6)
    
    local layout = Instance.new("UIListLayout")
    layout.SortOrder = Enum.SortOrder.LayoutOrder
    layout.Padding = UDim.new(0, 2)
    layout.Parent = scrollFrame
    
    local padding = Instance.new("UIPadding")
    padding.PaddingTop = UDim.new(0, 6)
    padding.PaddingBottom = UDim.new(0, 6)
    padding.PaddingLeft = UDim.new(0, 6)
    padding.PaddingRight = UDim.new(0, 6)
    padding.Parent = scrollFrame
    
    -- Variables
    local queue = {}
    local isExpanded = false
    
    -- Function to update layout orders (newest on top)
    local function updateLayoutOrders()
        local children = {}
        for _, child in pairs(scrollFrame:GetChildren()) do
            if child:IsA("Frame") and child ~= layout and child ~= padding then
                table.insert(children, child)
            end
        end
        
        for i = #children, 1, -1 do
            local orderIndex = (#children - i)
            children[i].LayoutOrder = orderIndex
        end
        
        local totalHeight = 0
        for _, child in ipairs(children) do
            totalHeight = totalHeight + child.Size.Y.Offset + 2
        end
        scrollFrame.CanvasSize = UDim2.new(0, 0, 0, totalHeight + 10)
        scrollFrame.CanvasPosition = Vector2.new(0, 0)
    end
    
    -- Add action (running animation then done)
    local function addAction(text)
        local row = Instance.new("Frame")
        row.Size = UDim2.new(1, 0, 0, 28)
        row.BackgroundColor3 = Color3.fromRGB(23, 23, 30)
        row.BorderSizePixel = 0
        row.Parent = scrollFrame
        
        local rowCorner = Instance.new("UICorner", row)
        rowCorner.CornerRadius = UDim.new(0, 4)
        
        -- Icon (spinner or check)
        local icon = Instance.new("TextLabel")
        icon.Size = UDim2.new(0, 25, 1, 0)
        icon.Position = UDim2.new(0, 5, 0, 0)
        icon.BackgroundTransparency = 1
        icon.TextXAlignment = Enum.TextXAlignment.Center
        icon.Font = Enum.Font.Gotham
        icon.TextSize = 12
        icon.Parent = row
        
        -- Action text
        local actionText = Instance.new("TextLabel")
        actionText.Text = text
        actionText.Size = UDim2.new(1, -40, 1, 0)
        actionText.Position = UDim2.new(0, 35, 0, 0)
        actionText.BackgroundTransparency = 1
        actionText.TextColor3 = Color3.fromRGB(215, 215, 235)
        actionText.TextXAlignment = Enum.TextXAlignment.Left
        actionText.Font = Enum.Font.Code
        actionText.TextSize = 11
        actionText.TextTruncate = Enum.TextTruncate.AtEnd
        actionText.Parent = row
        
        -- Start with spinner
        local spinFrames = {"◐", "◓", "◑", "◒"}
        local spinIndex = 1
        icon.Text = spinFrames[1]
        icon.TextColor3 = Color3.fromRGB(255, 210, 70)
        
        -- Spinner animation
        local spinnerConnection
        spinnerConnection = RunService.RenderStepped:Connect(function()
            if row and row.Parent then
                spinIndex = (spinIndex % 4) + 1
                icon.Text = spinFrames[spinIndex]
            else
                if spinnerConnection then
                    spinnerConnection:Disconnect()
                end
            end
        end)
        
        -- Store entry
        local entry = {
            row = row,
            icon = icon,
            actionText = actionText,
            spinnerConnection = spinnerConnection
        }
        
        table.insert(queue, entry)
        updateLayoutOrders()
        
        -- Function to mark as done
        local function markDone()
            if spinnerConnection then
                spinnerConnection:Disconnect()
            end
            if row and row.Parent and icon then
                icon.Text = "✓"
                icon.TextColor3 = Color3.fromRGB(80, 255, 120)
            end
        end
        
        return entry, markDone
    end
    
    -- Expand/collapse
    local function expand()
        isExpanded = true
        local tween = TweenService:Create(mainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad), {
            Size = UDim2.new(0, 450, 0, 400)
        })
        tween:Play()
        tween:Completed:Connect(function()
            expandedFrame.Visible = true
            minimizedButton.Visible = false
            updateLayoutOrders()
        end)
    end
    
    local function collapse()
        isExpanded = false
        expandedFrame.Visible = false
        minimizedButton.Visible = true
        local tween = TweenService:Create(mainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad), {
            Size = UDim2.new(0, 80, 0, 80)
        })
        tween:Play()
    end
    
    minimizedButton.MouseButton1Click:Connect(expand)
    if minimizedButton.TouchTap then
        minimizedButton.TouchTap:Connect(expand)
    end
    
    minimizeBtn.MouseButton1Click:Connect(collapse)
    if minimizeBtn.TouchTap then
        minimizeBtn.TouchTap:Connect(collapse)
    end
    
    -- Track tower placements
    local function trackPlacement(towerName, slot)
        if slot and slot > 0 then
            slotTowers[slot] = towerName
        end
    end
    
    -- Get slot from place method (handles different parameter orders)
    local function getSlotFromArgs(args)
        -- Different TDS libraries have different parameter orders
        -- Common patterns: Place(tower, x, y, z, slot) or Place(tower, slot, x, y, z)
        for i, arg in ipairs(args) do
            if type(arg) == "number" and arg > 0 and arg < 100 then
                -- Likely a slot number
                return arg
            end
        end
        return nil
    end
    
    -- WRAP TDS METHODS
    print("[TDS Overlay] Wrapping methods...")
    
    -- Wrap Place method
    if TDS.Place then
        local originalPlace = TDS.Place
        TDS.Place = function(self, ...)
            local args = {...}
            local towerName = args[1]
            local slot = getSlotFromArgs(args)
            
            -- Format display
            local x = type(args[2]) == "number" and args[2] or 0
            local z = type(args[4]) == "number" and args[4] or 0
            local displayText = string.format("Place: %s (%.0f, %.0f)", tostring(towerName), x, z)
            
            local entry, markDone = addAction(displayText)
            
            local success, result = pcall(originalPlace, self, ...)
            
            if success then
                if slot then
                    trackPlacement(towerName, slot)
                end
                markDone()
            else
                markDone()
                if entry and entry.icon then
                    entry.icon.Text = "✗"
                    entry.icon.TextColor3 = Color3.fromRGB(255, 90, 90)
                end
                warn("[TDS Overlay] Error in Place: " .. tostring(result))
            end
            
            return result
        end
    end
    
    -- Wrap Upgrade method
    if TDS.Upgrade then
        local originalUpgrade = TDS.Upgrade
        TDS.Upgrade = function(self, slot)
            local towerName = slotTowers[slot]
            local displayText
            if towerName then
                displayText = string.format("Upgrade: %s (slot %d)", towerName, slot)
            else
                displayText = string.format("Upgrade: slot %d", slot)
            end
            
            local entry, markDone = addAction(displayText)
            
            local success, result = pcall(originalUpgrade, self, slot)
            
            if success then
                markDone()
            else
                markDone()
                if entry and entry.icon then
                    entry.icon.Text = "✗"
                    entry.icon.TextColor3 = Color3.fromRGB(255, 90, 90)
                end
                warn("[TDS Overlay] Error in Upgrade: " .. tostring(result))
            end
            
            return result
        end
    end
    
    -- Wrap Sell method
    if TDS.Sell then
        local originalSell = TDS.Sell
        TDS.Sell = function(self, slot)
            local towerName = slotTowers[slot]
            local displayText
            if towerName then
                displayText = string.format("Sell: %s (slot %d)", towerName, slot)
            else
                displayText = string.format("Sell: slot %d", slot)
            end
            
            local entry, markDone = addAction(displayText)
            
            local success, result = pcall(originalSell, self, slot)
            
            if success then
                slotTowers[slot] = nil
                markDone()
            else
                markDone()
                if entry and entry.icon then
                    entry.icon.Text = "✗"
                    entry.icon.TextColor3 = Color3.fromRGB(255, 90, 90)
                end
                warn("[TDS Overlay] Error in Sell: " .. tostring(result))
            end
            
            return result
        end
    end
    
    -- Wrap Mode
    if TDS.Mode then
        local originalMode = TDS.Mode
        TDS.Mode = function(self, mode)
            local entry, markDone = addAction("Mode: " .. tostring(mode))
            
            local success, result = pcall(originalMode, self, mode)
            
            if success then
                markDone()
            else
                markDone()
                if entry and entry.icon then
                    entry.icon.Text = "✗"
                    entry.icon.TextColor3 = Color3.fromRGB(255, 90, 90)
                end
                warn("[TDS Overlay] Error in Mode: " .. tostring(result))
            end
            
            return result
        end
    end
    
    -- Wrap Loadout
    if TDS.Loadout then
        local originalLoadout = TDS.Loadout
        TDS.Loadout = function(self, ...)
            local args = {...}
            local displayText = "Loadout: " .. table.concat(args, ", ")
            if #displayText > 50 then
                displayText = displayText:sub(1, 47) .. "..."
            end
            
            local entry, markDone = addAction(displayText)
            
            local success, result = pcall(originalLoadout, self, ...)
            
            if success then
                markDone()
            else
                markDone()
                if entry and entry.icon then
                    entry.icon.Text = "✗"
                    entry.icon.TextColor3 = Color3.fromRGB(255, 90, 90)
                end
                warn("[TDS Overlay] Error in Loadout: " .. tostring(result))
            end
            
            return result
        end
    end
    
    -- Wrap Ready
    if TDS.Ready then
        local originalReady = TDS.Ready
        TDS.Ready = function(self)
            local entry, markDone = addAction("Ready - waiting for wave")
            
            local success, result = pcall(originalReady, self)
            
            if success then
                markDone()
            else
                markDone()
                if entry and entry.icon then
                    entry.icon.Text = "✗"
                    entry.icon.TextColor3 = Color3.fromRGB(255, 90, 90)
                end
                warn("[TDS Overlay] Error in Ready: " .. tostring(result))
            end
            
            return result
        end
    end
    
    -- Wrap Ability
    if TDS.Ability then
        local originalAbility = TDS.Ability
        TDS.Ability = function(self, slot, abilityName)
            local towerName = slotTowers[slot]
            local displayText
            if towerName then
                displayText = string.format("Ability: %s (%s)", towerName, tostring(abilityName))
            else
                displayText = string.format("Ability: slot %d - %s", slot, tostring(abilityName))
            end
            
            local entry, markDone = addAction(displayText)
            
            local success, result = pcall(originalAbility, self, slot, abilityName)
            
            if success then
                markDone()
            else
                markDone()
                if entry and entry.icon then
                    entry.icon.Text = "✗"
                    entry.icon.TextColor3 = Color3.fromRGB(255, 90, 90)
                end
                warn("[TDS Overlay] Error in Ability: " .. tostring(result))
            end
            
            return result
        end
    end
    
    -- Wrap SetOption
    if TDS.SetOption then
        local originalSetOption = TDS.SetOption
        TDS.SetOption = function(self, slot, key, value)
            local displayText = string.format("SetOption: slot %d - %s = %s", slot or 0, tostring(key), tostring(value))
            
            local entry, markDone = addAction(displayText)
            
            local success, result = pcall(originalSetOption, self, slot, key, value)
            
            if success then
                markDone()
            else
                markDone()
                if entry and entry.icon then
                    entry.icon.Text = "✗"
                    entry.icon.TextColor3 = Color3.fromRGB(255, 90, 90)
                end
                warn("[TDS Overlay] Error in SetOption: " .. tostring(result))
            end
            
            return result
        end
    end
    
    print("[TDS Overlay] Successfully attached!")
    print("[TDS Overlay] Window starts as minimized cube in top-left corner")
    print("[TDS Overlay] Click 'Debug' to expand and see actions")
    
    return TDS
end

return attachOverlay
