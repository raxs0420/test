-- Try to find TDS in different places
local TDS = nil

-- Check if TDS is in _G
if _G.TDS then
    TDS = _G.TDS
    print("[TDS Overlay] Found TDS in _G")
end

-- Check if TDS is a global variable
if not TDS and getfenv().TDS then
    TDS = getfenv().TDS
    print("[TDS Overlay] Found TDS as global variable")
end

-- Check for any table with TDS methods
if not TDS then
    for k, v in pairs(getfenv()) do
        if type(v) == "table" and (v.Mode or v.Place or v.Upgrade or v.Sell or v.Loadout) then
            print("[TDS Overlay] Found TDS-like table:", k)
            TDS = v
            break
        end
    end
end

-- If still not found, show error
if not TDS then

-- Drag functionality
local dragging = false
local dragStart, startPos

titleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = mainFrame.Position
    end
end)

UIS.InputChanged:Connect(function(input)
    if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - dragStart
        mainFrame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)

UIS.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)

-- Function to add an action

    actionCount = actionCount + 1
    
    local row = Instance.new("Frame")
    row.Size = UDim2.new(1, 0, 0, 30)
    row.BackgroundColor3 = Color3.fromRGB(25, 25, 32)
    row.BorderSizePixel = 0
    row.Parent = scrollFrame
    
    local icon = Instance.new("TextLabel")
    icon.Size = UDim2.new(0, 25, 1, 0)
    icon.Position = UDim2.new(0, 5, 0, 0)
    icon.BackgroundTransparency = 1
    icon.TextXAlignment = Enum.TextXAlignment.Center
    icon.Font = Enum.Font.Gotham
    icon.TextSize = 14
    icon.Parent = row
    
    local actionText = Instance.new("TextLabel")
    actionText.Text = text
    actionText.Size = UDim2.new(1, -120, 1, 0)
    actionText.Position = UDim2.new(0, 35, 0, 0)
    actionText.BackgroundTransparency = 1
    actionText.TextColor3 = Color3.fromRGB(210, 210, 230)
    actionText.TextXAlignment = Enum.TextXAlignment.Left
    actionText.Font = Enum.Font.Code
    actionText.TextSize = 10
    actionText.TextTruncate = Enum.TextTruncate.AtEnd
    actionText.Parent = row
    
    local statusText = Instance.new("TextLabel")
    statusText.Size = UDim2.new(0, 70, 1, 0)
    statusText.Position = UDim2.new(1, -75, 0, 0)
    statusText.BackgroundTransparency = 1
    statusText.TextXAlignment = Enum.TextXAlignment.Right
    statusText.Font = Enum.Font.GothamBold
    statusText.TextSize = 9
    statusText.Parent = row
    
    if status == "running" then
        icon.Text = "●"
        icon.TextColor3 = Color3.fromRGB(255, 200, 50)
        statusText.Text = "RUNNING"
        statusText.TextColor3 = Color3.fromRGB(255, 200, 50)
    elseif status == "done" then
        icon.Text = "✓"
        icon.TextColor3 = Color3.fromRGB(100, 255, 100)
        statusText.Text = "DONE"
        statusText.TextColor3 = Color3.fromRGB(100, 255, 100)
        completedCount = completedCount + 1
    elseif status == "error" then
        icon.Text = "✗"
        icon.TextColor3 = Color3.fromRGB(255, 80, 80)
        statusText.Text = "ERROR"
        statusText.TextColor3 = Color3.fromRGB(255, 80, 80)
    else
        icon.Text = "○"
        icon.TextColor3 = Color3.fromRGB(255, 150, 100)
        statusText.Text = "PENDING"
        statusText.TextColor3 = Color3.fromRGB(255, 150, 100)
    end
    
    counterText.Text = completedCount .. " / " .. actionCount
    
    -- Auto-scroll
    task.defer(function()
        scrollFrame.CanvasPosition = Vector2.new(0, scrollFrame.CanvasSize.Y.Offset)
    end)
    
    return {row = row, icon = icon, statusText = statusText, actionText = actionText}
end

-- Function to update status
local function updateStatus(entry, newStatus)
    if newStatus == "done" then
        entry.icon.Text = "✓"
        entry.icon.TextColor3 = Color3.fromRGB(100, 255, 100)
        entry.statusText.Text = "DONE"
        entry.statusText.TextColor3 = Color3.fromRGB(100, 255, 100)
        completedCount = completedCount + 1
        counterText.Text = completedCount .. " / " .. actionCount
    elseif newStatus == "error" then
        entry.icon.Text = "✗"
        entry.icon.TextColor3 = Color3.fromRGB(255, 80, 80)
        entry.statusText.Text = "ERROR"
        entry.statusText.TextColor3 = Color3.fromRGB(255, 80, 80)
        entry.actionText.TextColor3 = Color3.fromRGB(255, 150, 150)
    end
end

-- WRAP TDS METHODS
print("[TDS Overlay] Wrapping TDS methods...")

local wrappedCount = 0
for methodName, methodFunc in pairs(TDS) do
    if type(methodFunc) == "function" then
        print("[TDS Overlay] Wrapping:", methodName)
        
        local originalMethod = methodFunc
        TDS[methodName] = function(self, ...)
            local args = {...}
            
            -- Create label
            local label = methodName
            if #args > 0 then
                local parts = {}
                for i, arg in ipairs(args) do
                    if type(arg) == "string" then
                        parts[i] = '"' .. arg .. '"'
                    elseif type(arg) == "number" then
                        parts[i] = string.format("%g", arg)
                    else
                        parts[i] = tostring(arg)
                    end
                end
                label = label .. "(" .. table.concat(parts, ", ") .. ")"
            else
                label = label .. "()"
            end
            
            -- Truncate if needed
            if #label > 60 then
                label = label:sub(1, 57) .. "..."
            end
            
            local entry = addAction(label, "running")
            
            local success, result = pcall(originalMethod, self, table.unpack(args))
            
            if success then
                updateStatus(entry, "done")
            else
                updateStatus(entry, "error")
                warn("[Overlay] Error in " .. methodName .. ": " .. tostring(result))
            end
            
            return result
        end
        
        wrappedCount = wrappedCount + 1
    end
end

print("[TDS Overlay] Wrapped", wrappedCount, "methods")

-- Add welcome message
addAction("✓ Overlay active - monitoring " .. wrappedCount .. " methods", "done")
