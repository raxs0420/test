-- // TDS Debug Overlay - PROPER VERSION

local function attachOverlay(TDS)
    print("[DEBUG] attachOverlay called")
    print("[DEBUG] TDS type:", type(TDS))
    
    if not TDS then 
        warn("[Overlay] TDS table is nil")
        return 
    end
    
    -- List ALL methods in your TDS table
    print("[DEBUG] TDS methods found:")
    for k, v in pairs(TDS) do
        if type(v) == "function" then
            print("[DEBUG]   -", k)
        end
    end
    
    local Players = game:GetService("Players")
    local UIS = game:GetService("UserInputService")
    local localPlayer = Players.LocalPlayer
    local playerGui = localPlayer:WaitForChild("PlayerGui")
    
    -- Remove old overlay
    local existingOverlay = playerGui:FindFirstChild("TDS_DebugOverlay")
    if existingOverlay then
        existingOverlay:Destroy()
        wait(0.1)
    end
    
    -- Create main GUI
    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "TDS_DebugOverlay"
    screenGui.ResetOnSpawn = false
    screenGui.DisplayOrder = 999
    screenGui.IgnoreGuiInset = true
    screenGui.Parent = playerGui
    
    local mainFrame = Instance.new("Frame")
    mainFrame.Size = UDim2.new(0, 450, 0, 400)
    mainFrame.Position = UDim2.new(0.5, -225, 0.2, 0)
    mainFrame.BackgroundColor3 = Color3.fromRGB(20, 20, 25)
    mainFrame.BorderSizePixel = 1
    mainFrame.BorderColor3 = Color3.fromRGB(60, 60, 80)
    mainFrame.Parent = screenGui
    
    -- Corner radius
    local corner = Instance.new("UICorner", mainFrame)
    corner.CornerRadius = UDim.new(0, 8)
    
    -- Title bar
    local titleBar = Instance.new("Frame")
    titleBar.Size = UDim2.new(1, 0, 0, 35)
    titleBar.BackgroundColor3 = Color3.fromRGB(30, 30, 38)
    titleBar.BorderSizePixel = 0
    titleBar.Parent = mainFrame
    
    local titleCorner = Instance.new("UICorner", titleBar)
    titleCorner.CornerRadius = UDim.new(0, 8)
    
    local titleText = Instance.new("TextLabel")
    titleText.Text = "⚙ TDS Debug Monitor"
    titleText.Size = UDim2.new(1, -80, 1, 0)
    titleText.Position = UDim2.new(0, 10, 0, 0)
    titleText.BackgroundTransparency = 1
    titleText.TextColor3 = Color3.fromRGB(220, 220, 240)
    titleText.TextXAlignment = Enum.TextXAlignment.Left
    titleText.Font = Enum.Font.GothamBold
    titleText.TextSize = 13
    titleText.Parent = titleBar
    
    local counterText = Instance.new("TextLabel")
    counterText.Text = "0 / 0"
    counterText.Size = UDim2.new(0, 70, 1, 0)
    counterText.Position = UDim2.new(1, -80, 0, 0)
    counterText.BackgroundTransparency = 1
    counterText.TextColor3 = Color3.fromRGB(150, 150, 170)
    counterText.TextXAlignment = Enum.TextXAlignment.Right
    counterText.Font = Enum.Font.Gotham
    counterText.TextSize = 11
    counterText.Parent = titleBar
    
    -- Close button
    local closeBtn = Instance.new("TextButton")
    closeBtn.Text = "✕"
    closeBtn.Size = UDim2.new(0, 25, 0, 25)
    closeBtn.Position = UDim2.new(1, -30, 0, 5)
    closeBtn.BackgroundColor3 = Color3.fromRGB(50, 40, 40)
    closeBtn.TextColor3 = Color3.fromRGB(200, 100, 100)
    closeBtn.Font = Enum.Font.GothamBold
    closeBtn.TextSize = 14
    closeBtn.BorderSizePixel = 0
    closeBtn.Parent = titleBar
    
    local closeCorner = Instance.new("UICorner", closeBtn)
    closeCorner.CornerRadius = UDim.new(0, 4)
    
    closeBtn.MouseButton1Click:Connect(function()
        screenGui:Destroy()
    end)
    
    -- Minimize button
    local minBtn = Instance.new("TextButton")
    minBtn.Text = "−"
    minBtn.Size = UDim2.new(0, 25, 0, 25)
    minBtn.Position = UDim2.new(1, -60, 0, 5)
    minBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
    minBtn.TextColor3 = Color3.fromRGB(180, 180, 200)
    minBtn.Font = Enum.Font.GothamBold
    minBtn.TextSize = 18
    minBtn.BorderSizePixel = 0
    minBtn.Parent = titleBar
    
    local minCorner = Instance.new("UICorner", minBtn)
    minCorner.CornerRadius = UDim.new(0, 4)
    
    -- Scroll frame for actions
    local scrollFrame = Instance.new("ScrollingFrame")
    scrollFrame.Size = UDim2.new(1, -10, 1, -45)
    scrollFrame.Position = UDim2.new(0, 5, 0, 40)
    scrollFrame.BackgroundColor3 = Color3.fromRGB(15, 15, 20)
    scrollFrame.BorderSizePixel = 0
    scrollFrame.CanvasSize = UDim2.new(0, 0, 0, 0)
    scrollFrame.AutomaticCanvasSize = Enum.AutomaticSize.Y
    scrollFrame.ScrollBarThickness = 5
    scrollFrame.ScrollBarImageColor3 = Color3.fromRGB(80, 80, 100)
    scrollFrame.Parent = mainFrame
    
    local scrollCorner = Instance.new("UICorner", scrollFrame)
    scrollCorner.CornerRadius = UDim.new(0, 6)
    
    local layout = Instance.new("UIListLayout")
    layout.Padding = UDim.new(0, 2)
    layout.Parent = scrollFrame
    
    local padding = Instance.new("UIPadding")
    padding.PaddingTop = UDim.new(0, 5)
    padding.PaddingBottom = UDim.new(0, 5)
    padding.PaddingLeft = UDim.new(0, 5)
    padding.PaddingRight = UDim.new(0, 5)
    padding.Parent = scrollFrame
    
    local queue = {}
    local minimized = false
    
    minBtn.MouseButton1Click:Connect(function()
        minimized = not minimized
        if minimized then
            mainFrame.Size = UDim2.new(0, 450, 0, 40)
            scrollFrame.Visible = false
            minBtn.Text = "+"
        else
            mainFrame.Size = UDim2.new(0, 450, 0, 400)
            scrollFrame.Visible = true
            minBtn.Text = "−"
        end
    end)
    
    -- Drag functionality
    local dragging = false
    local dragStart, startPos
    
    titleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPos = mainFrame.Position
        end
    end)
    
    UIS.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - dragStart
            mainFrame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
    
    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
    
    -- Add action row
    local function addAction(text, statusType)
        local row = Instance.new("Frame")
        row.Size = UDim2.new(1, 0, 0, 28)
        row.BackgroundColor3 = Color3.fromRGB(25, 25, 32)
        row.BorderSizePixel = 0
        row.Parent = scrollFrame
        
        local rowCorner = Instance.new("UICorner", row)
        rowCorner.CornerRadius = UDim.new(0, 4)
        
        local timestamp = Instance.new("TextLabel")
        timestamp.Text = os.date("%H:%M:%S")
        timestamp.Size = UDim2.new(0, 55, 1, 0)
        timestamp.Position = UDim2.new(0, 5, 0, 0)
        timestamp.BackgroundTransparency = 1
        timestamp.TextColor3 = Color3.fromRGB(120, 120, 140)
        timestamp.TextXAlignment = Enum.TextXAlignment.Left
        timestamp.Font = Enum.Font.Code
        timestamp.TextSize = 9
        timestamp.Parent = row
        
        local actionText = Instance.new("TextLabel")
        actionText.Text = text
        actionText.Size = UDim2.new(1, -130, 1, 0)
        actionText.Position = UDim2.new(0, 65, 0, 0)
        actionText.BackgroundTransparency = 1
        actionText.TextColor3 = Color3.fromRGB(210, 210, 230)
        actionText.TextXAlignment = Enum.TextXAlignment.Left
        actionText.Font = Enum.Font.Code
        actionText.TextSize = 11
        actionText.TextTruncate = Enum.TextTruncate.AtEnd
        actionText.Parent = row
        
        local statusLabel = Instance.new("TextLabel")
        
        if statusType == "pending" then
            statusLabel.Text = "⏳ PENDING"
            statusLabel.TextColor3 = Color3.fromRGB(255, 150, 100)
        elseif statusType == "running" then
            statusLabel.Text = "🔄 RUNNING"
            statusLabel.TextColor3 = Color3.fromRGB(255, 200, 50)
        elseif statusType == "done" then
            statusLabel.Text = "✅ DONE"
            statusLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
        elseif statusType == "error" then
            statusLabel.Text = "❌ ERROR"
            statusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
        else
            statusLabel.Text = "⚡ ACTIVE"
            statusLabel.TextColor3 = Color3.fromRGB(80, 150, 255)
        end
        
        statusLabel.Size = UDim2.new(0, 75, 1, 0)
        statusLabel.Position = UDim2.new(1, -80, 0, 0)
        statusLabel.BackgroundTransparency = 1
        statusLabel.TextXAlignment = Enum.TextXAlignment.Right
        statusLabel.Font = Enum.Font.GothamBold
        statusLabel.TextSize = 10
        statusLabel.Parent = row
        
        local entry = {
            row = row, 
            actionText = actionText, 
            statusLabel = statusLabel, 
            status = statusType
        }
        table.insert(queue, entry)
        
        -- Update counter
        local total = #queue
        local done = 0
        for _, e in ipairs(queue) do
            if e.status == "done" or e.status == "active" then
                done = done + 1
            end
        end
        counterText.Text = done .. " / " .. total
        
        -- Auto-scroll
        task.defer(function()
            scrollFrame.CanvasPosition = Vector2.new(0, scrollFrame.CanvasSize.Y.Offset)
        end)
        
        return entry
    end
    
    -- Update status function
    local function updateStatus(entry, newStatus)
        entry.status = newStatus
        if newStatus == "done" then
            entry.statusLabel.Text = "✅ DONE"
            entry.statusLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
        elseif newStatus == "error" then
            entry.statusLabel.Text = "❌ ERROR"
            entry.statusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
            entry.actionText.TextColor3 = Color3.fromRGB(255, 150, 150)
        elseif newStatus == "running" then
            entry.statusLabel.Text = "🔄 RUNNING"
            entry.statusLabel.TextColor3 = Color3.fromRGB(255, 200, 50)
        elseif newStatus == "active" then
            entry.statusLabel.Text = "⚡ ACTIVE"
            entry.statusLabel.TextColor3 = Color3.fromRGB(80, 150, 255)
        end
        
        -- Update counter
        local total = #queue
        local done = 0
        for _, e in ipairs(queue) do
            if e.status == "done" or e.status == "active" then
                done = done + 1
            end
        end
        counterText.Text = done .. " / " .. total
    end
    
    -- WRAP YOUR ACTUAL TDS METHODS (NO MOCKS)
    print("[DEBUG] Wrapping TDS methods...")
    
    -- Get all methods from your TDS table
    local wrappedCount = 0
    for methodName, methodFunc in pairs(TDS) do
        if type(methodFunc) == "function" then
            print("[DEBUG] Wrapping method:", methodName)
            
            -- Create a wrapper that preserves the original method
            local originalMethod = methodFunc
            TDS[methodName] = function(self, ...)
                local args = {...}
                
                -- Create a descriptive label
                local label = methodName .. "("
                local argStrings = {}
                for i, arg in ipairs(args) do
                    if type(arg) == "string" then
                        argStrings[i] = '"' .. arg .. '"'
                    elseif type(arg) == "number" then
                        argStrings[i] = string.format("%.1f", arg)
                    else
                        argStrings[i] = tostring(arg)
                    end
                end
                label = label .. table.concat(argStrings, ", ") .. ")"
                
                -- Truncate if too long
                if #label > 60 then
                    label = label:sub(1, 57) .. "..."
                end
                
                -- Add to queue
                local entry = addAction(label, "running")
                
                -- Call original method
                local success, result = pcall(originalMethod, self, table.unpack(args))
                
                if success then
                    updateStatus(entry, "done")
                else
                    updateStatus(entry, "error")
                    warn("[Overlay] Error in " .. methodName .. ": " .. tostring(result))
                end
                
                return result
            end
            
            wrappedCount = wrappedCount + 1
        end
    end
    
    print("[DEBUG] Wrapped", wrappedCount, "methods")
    
    -- Add welcome message
    addAction("✓ Overlay active - monitoring " .. wrappedCount .. " methods", "done")
    addAction("→ Ready for TDS calls", "pending")
    
    -- Add a test to show it's working
    task.spawn(function()
        wait(0.5)
        print("[DEBUG] Overlay ready! Any TDS calls will now appear here")
    end)
    
    return TDS
end

-- MAIN EXECUTION
print("=" .. string.rep("=", 50))
print("TDS Debug Overlay - Loading...")
print("=" .. string.rep("=", 50))

-- Find TDS - check common locations
local tdsInstance = nil

-- Check if TDS is in _G
if _G.TDS then
    print("[INFO] Found TDS in _G")
    tdsInstance = _G.TDS
end

-- Check if TDS is a global variable
if not tdsInstance and TDS then
    print("[INFO] Found TDS as global variable")
    tdsInstance = TDS
end

-- Check if there's a function that returns TDS (common pattern)
if not tdsInstance then
    print("[INFO] Looking for TDS in _G.TDS or global TDS variable")
    for k, v in pairs(getfenv()) do
        if type(v) == "table" and (v.Mode or v.Place or v.Upgrade) then
            print("[INFO] Found potential TDS table in:", k)
            tdsInstance = v
            break
        end
    end
end

if tdsInstance then
    print("[INFO] Attaching overlay to existing TDS...")
    attachOverlay(tdsInstance)
    print("[INFO] Overlay attached successfully!")
    print("[INFO] Your TDS calls will now appear in the debug window")
else
    print("[ERROR] No TDS instance found!")
    print("[ERROR] Make sure TDS is loaded BEFORE this script")
    print("[ERROR] Example usage:")
    print('  local TDS = loadstring(game:HttpGet("your_tds_url"))()')
    print('  loadstring(game:HttpGet("this_script_url"))()(TDS)')
    
    -- Create a visible error message
    local player = game.Players.LocalPlayer
    if player and player.PlayerGui then
        local errorGui = Instance.new("ScreenGui")
        errorGui.Name = "TDS_Error"
        errorGui.Parent = player.PlayerGui
        
        local errorFrame = Instance.new("Frame")
        errorFrame.Size = UDim2.new(0, 400, 0, 150)
        errorFrame.Position = UDim2.new(0.5, -200, 0.5, -75)
        errorFrame.BackgroundColor3 = Color3.fromRGB(40, 20, 20)
        errorFrame.BorderSizePixel = 2
        errorFrame.BorderColor3 = Color3.fromRGB(200, 50, 50)
        errorFrame.Parent = errorGui
        
        local errorText = Instance.new("TextLabel")
        errorText.Text = "ERROR: No TDS instance found!\n\nMake sure TDS is loaded BEFORE this script\n\nExample:\nlocal TDS = loadstring(game:HttpGet(...))()\nloadstring(game:HttpGet(...))()(TDS)"
        errorText.Size = UDim2.new(1, -20, 1, -20)
        errorText.Position = UDim2.new(0, 10, 0, 10)
        errorText.BackgroundTransparency = 1
        errorText.TextColor3 = Color3.fromRGB(255, 200, 200)
        errorText.TextWrapped = true
        errorText.Font = Enum.Font.Gotham
        errorText.TextSize = 12
        errorText.Parent = errorFrame
        
        local closeError = Instance.new("TextButton")
        closeError.Text = "OK"
        closeError.Size = UDim2.new(0, 60, 0, 30)
        closeError.Position = UDim2.new(0.5, -30, 1, -40)
        closeError.BackgroundColor3 = Color3.fromRGB(100, 50, 50)
        closeError.TextColor3 = Color3.fromRGB(255, 255, 255)
        closeError.Parent = errorFrame
        closeError.MouseButton1Click:Connect(function()
            errorGui:Destroy()
        end)
    end
end
