            slotButtons[i].BackgroundColor3 = Color3.fromRGB(100, 100, 0)

            slotLabels[i].Text = "[Empty]"

local function updateSwitcherStatus(message, color)
    switcherStatus.Text = "Selected: " .. #selectedTowers .. "/5\n" .. message
    switcherStatus.TextColor3 = color or Color3.fromRGB(200, 200, 200)
end

local function createSwitcherTowerButton(towerName, index, isEven)
    local towerBtn = Instance.new("TextButton")
    towerBtn.Name = "TowerBtn_" .. towerName
    towerBtn.Size = UDim2.new(1, -8, 0, 26)
    towerBtn.Position = UDim2.new(0, 4, 0, (index-1) * 28)
    towerBtn.BackgroundColor3 = isEven and Color3.fromRGB(40, 40, 55) or Color3.fromRGB(35, 35, 50)
    towerBtn.Text = towerName
    towerBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    towerBtn.TextSize = 11
    towerBtn.Font = Enum.Font.SourceSans
    towerBtn.AutoButtonColor = false
    towerBtn.Parent = switcherTowersScroll
    
    Instance.new("UICorner", towerBtn).CornerRadius = UDim.new(0, 4)
    
    local originalColor = towerBtn.BackgroundColor3
    switcherTowerButtons[towerName] = towerBtn
    
    for _, selectedTower in ipairs(selectedTowers) do
        if selectedTower == towerName then
            towerBtn.BackgroundColor3 = Color3.fromRGB(70, 130, 70)
            break
        end
    end
    
    towerBtn.MouseButton1Click:Connect(function()
        local alreadySelected = false
        local selectedIndex = 0
        
        for i, selectedTower in ipairs(selectedTowers) do
            if selectedTower == towerName then
                alreadySelected = true
                selectedIndex = i
                break
            end
        end
        
        if alreadySelected then
            table.remove(selectedTowers, selectedIndex)
            towerBtn.BackgroundColor3 = originalColor
            updateSwitcherStatus(towerName .. " removed", Color3.fromRGB(255, 150, 150))
        else
            if #selectedTowers >= 5 then
                updateSwitcherStatus("Max 5 reached! Remove some first", Color3.fromRGB(255, 100, 100))
                towerBtn.BackgroundColor3 = Color3.fromRGB(255, 80, 80)
                task.wait(0.5)
                towerBtn.BackgroundColor3 = originalColor
                return
            end
            
            table.insert(selectedTowers, towerName)
            towerBtn.BackgroundColor3 = Color3.fromRGB(70, 130, 70)
            updateSwitcherStatus(towerName .. " added", Color3.fromRGB(100, 255, 100))
        end
        
        updateSwitcherCountDisplay()
    end)
    
    towerBtn.MouseEnter:Connect(function()
        if towerBtn.BackgroundColor3 ~= Color3.fromRGB(70, 130, 70) then
            towerBtn.BackgroundColor3 = Color3.fromRGB(55, 55, 70)
        end
    end)
    
    towerBtn.MouseLeave:Connect(function()
        local isSelected = false
        for _, selectedTower in ipairs(selectedTowers) do
            if selectedTower == towerName then
                isSelected = true
                break
            end
        end
        
        towerBtn.BackgroundColor3 = isSelected and Color3.fromRGB(70, 130, 70) or originalColor
    end)
end

local function populateTowerList(searchText)
    for _, child in ipairs(scrollFrame:GetChildren()) do
        if child:IsA("TextButton") then
            child:Destroy()
        end
    end
    
    local filteredTowers = {}
    if searchText and searchText ~= "" then
        searchText = searchText:lower()
        for _, tower in ipairs(allTowers) do
            if tower:lower():find(searchText) then
                table.insert(filteredTowers, tower)
            end
        end
    else
        filteredTowers = allTowers
    end
    
    for i, towerName in ipairs(filteredTowers) do
        local button = Instance.new("TextButton")
        button.Size = UDim2.new(1, -4, 0, 28)
        button.Position = UDim2.new(0, 2, 0, (i-1) * 30)
        button.BackgroundColor3 = i % 2 == 0 and Color3.fromRGB(40, 40, 55) or Color3.fromRGB(35, 35, 50)
        button.Text = towerName
        button.TextColor3 = Color3.fromRGB(255, 255, 255)
        button.TextSize = 12
        button.Font = Enum.Font.SourceSans
        button.AutoButtonColor = false
        button.Parent = scrollFrame
        
        Instance.new("UICorner", button).CornerRadius = UDim.new(0, 4)
        
        local originalColor = button.BackgroundColor3
        
        button.MouseButton1Click:Connect(function()
            if lastHotbarSelected then
                HotbarMapping[lastHotbarSelected] = towerName
                stackerStatus.Text = towerName .. " → Slot " .. lastHotbarSelected
                updateSlotHighlights()
            else
                stackerStatus.Text = "Click slot button first!"
            end
        end)
        
        button.MouseEnter:Connect(function()
            if button.BackgroundColor3 == originalColor then
                button.BackgroundColor3 = Color3.fromRGB(55, 55, 70)
            end
        end)
        
        button.MouseLeave:Connect(function()
            if button.BackgroundColor3 == Color3.fromRGB(55, 55, 70) then
                button.BackgroundColor3 = originalColor
            end
        end)
    end
    
    scrollFrame.CanvasSize = UDim2.new(0, 0, 0, #filteredTowers * 30)
end

local function getWorldPosition(screenPosition)
    local ray = camera:ScreenPointToRay(screenPosition.X, screenPosition.Y)
    
    local distance = -ray.Origin.Y / ray.Direction.Y
    if distance > 0 then
        local worldPoint = ray.Origin + (ray.Direction * distance)
        return worldPoint.X, worldPoint.Z
    end
    
    return nil, nil
end

local lastCameraCFrame = camera.CFrame
local lastCameraMoveTime = tick()
local cameraCooldown = 0.3

game:GetService("RunService").RenderStepped:Connect(function()
    local currentCFrame = camera.CFrame
    
    if UIS.TouchEnabled then
        if (currentCFrame.Position - lastCameraCFrame.Position).Magnitude > 0.01 or
           currentCFrame.LookVector:Dot(lastCameraCFrame.LookVector) < 0.9999 then
            lastCameraMoveTime = tick()
        end
    else
        if (currentCFrame.Position - lastCameraCFrame.Position).Magnitude > 0.001 then
            lastCameraMoveTime = tick()
        end
    end
    
    lastCameraCFrame = currentCFrame
end)

local function canPlace()
    return tick() - lastCameraMoveTime > cameraCooldown
end

local mt = getrawmetatable(game)
setreadonly(mt, false)
local oldNamecall = mt.__namecall

mt.__namecall = function(self, ...)
    local method = getnamecallmethod()
    local args = {...}

    if method == "FireServer" and self == RemoteEvent then
        if args[1] == "Hotbar" and args[2] == "Click" then
            lastHotbarSelected = args[3]
            if currentTab == "stacker" then
                stackerStatus.Text = "Selected slot: " .. lastHotbarSelected
            end
            updateSlotHighlights()
        end
    end

    return oldNamecall(self, ...)
end

setreadonly(mt, true)

UIS.InputBegan:Connect(function(input, processed)
    if processed then return end
    
    if input.UserInputType == Enum.UserInputType.MouseButton2 then
        if lastHotbarSelected and HotbarMapping[lastHotbarSelected] and canPlace() and isPlacingActive then
            local mousePos = UIS:GetMouseLocation()
            local x, z = getWorldPosition(mousePos)
            if x and z then
                stackerStatus.Text = "Placing " .. HotbarMapping[lastHotbarSelected]
                
                local success, err = pcall(function()
                    RemoteFunction:InvokeServer("Troops", "Pl\208\176ce", {
                        Rotation = CFrame.Angles(0, 0, 0),
                        Position = Vector3.new(x, 80, z)
                    }, HotbarMapping[lastHotbarSelected])
                end)
                
                if success then
                    stackerStatus.Text = "Placed " .. HotbarMapping[lastHotbarSelected] .. " from slot " .. lastHotbarSelected
                else
                    stackerStatus.Text = "Error: " .. tostring(err)
                end
            end
        elseif not isPlacingActive then
            stackerStatus.Text = "Placing is disabled!"
        end
    end
    
    if input.UserInputType == Enum.UserInputType.MouseButton1 and not UIS.TouchEnabled then
        if not mainFrame.Visible then
            lastHotbarSelected = nil
            if currentTab == "stacker" then
                stackerStatus.Text = "Click slot button first"
            end
            updateSlotHighlights()
        end
    end
end)

if UIS.TouchEnabled then
    local activeTouch = nil
    local touchStartTime = 0
    local touchStartPosition = nil
    
    UIS.InputBegan:Connect(function(input, processed)
        if processed then return end
        
        if input.UserInputType == Enum.UserInputType.Touch then
            if lastHotbarSelected and HotbarMapping[lastHotbarSelected] and canPlace() and isPlacingActive then
                activeTouch = input
                touchStartTime = tick()
                touchStartPosition = input.Position
                stackerStatus.Text = "Placing " .. HotbarMapping[lastHotbarSelected]
            elseif not isPlacingActive then
                stackerStatus.Text = "Placing is disabled!"
            end
        end
    end)
    
    UIS.InputEnded:Connect(function(input, processed)
        if input.UserInputType == Enum.UserInputType.Touch and input == activeTouch then
            activeTouch = nil
        end
    end)
    
    while true do
        if activeTouch and touchStartPosition and canPlace() and isPlacingActive then
            if tick() - touchStartTime >= 0.4 then
                if lastHotbarSelected and HotbarMapping[lastHotbarSelected] then
                    local x, z = getWorldPosition(touchStartPosition)
                    if x and z then
                        RemoteFunction:InvokeServer("Troops", "Pl\208\176ce", {
                            Rotation = CFrame.Angles(0, 0, 0),
                            Position = Vector3.new(x, 80, z)
                        }, HotbarMapping[lastHotbarSelected])
                        stackerStatus.Text = "Placed " .. HotbarMapping[lastHotbarSelected] .. " from slot " .. lastHotbarSelected
                        activeTouch = nil
                    end
                end
            end
        end
        
        wait(0.1)
    end
end

        end
    end
end)
